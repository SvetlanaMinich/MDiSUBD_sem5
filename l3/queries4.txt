TABLE1:
CREATE TABLE Products
(
    Id SERIAL PRIMARY KEY,
    ProductName VARCHAR(30) NOT NULL,
    Company VARCHAR(20) NOT NULL,
    ProductCount INTEGER DEFAULT 0,
    Price NUMERIC NOT NULL
);

TABLE2:
CREATE TABLE Customers
(
    Id SERIAL PRIMARY KEY,
    FirstName VARCHAR(30) NOT NULL
);

TABLE3:
CREATE TABLE Orders
(
    Id SERIAL PRIMARY KEY,
    ProductId INTEGER NOT NULL REFERENCES Products(Id) ON DELETE CASCADE,
    CustomerId INTEGER NOT NULL REFERENCES Customers(Id) ON DELETE CASCADE,
    CreatedAt DATE NOT NULL,
    ProductCount INTEGER DEFAULT 1,
    Price NUMERIC NOT NULL
);

список всех полей с рейтингом выше 4, доступных для бронирования в заданном диапазоне дат, по цене ниже 500

select F.field_name, F.rating, F.price_per_hour, S.time_from, S.time_to, S.available
from Field as F
join Schedule as S on S.field_id = F.Id
where S.available = true and
      S.time_from >= "" and
      S.time_to <= "" and
      F.rating > 4 and
      F.price_per_hour < 500;


список клиентов, которые оставили отзывы с рейтингом ниже среднего рейтинга всех полей
select C.name, C.surname, R.text, R.rating
from Client as C
join Review as R on R.client_id = C.Id
where R.rating < (
    select avr(F.rating)
    from Field as F
);


Получить поля с наибольшим количеством бронирований в текущем месяце
select f.field_name, count(r.reservation_id) as reservation_count
from field as f
join schedule as s on s.field_id = f.field_id
join reservation as r on r.schedule_id = s.schedule_id
where 
	r.created_at >= date_trunc('month', CURRENT_DATE) 
    AND r.created_at < date_trunc('month', CURRENT_DATE) + interval '1 month'
group by f.field_name, f.field_id 
order by reservation_count desc
limit 1;
