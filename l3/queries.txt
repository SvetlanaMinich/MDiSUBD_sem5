create database app;
drop database app;

create table tapp (
    id serial primary key,
    nickname varchar(50) unique check(nickname != ''),
    age integer not null constraint age_limit check(age >= 18 and age < 120),
    registered_date date not null default now(),
    is_online boolean not null,
);

create table torder (
    id serial primary key,
    price decimal(8,2) not null constraint price_not_zero check(price > 0),
    action_date timestamp default now(),
    customer_id integer references tapp (id) on delete set null,
);

alter table tapp
add column password varchar(12) not null check(length(password) >= 8),
drop column age,
alter column nickname set check(length(nickname) > 1)
rename column registered_date to r_date,
